// Vigie v2 – serveur relais (Cloudflare Worker)
// Rôle : clé API secrète, transcription des notes vocales, analyse IA, registre (D1), photos et audio (KV).
//
// Variables (Settings > Variables and Secrets) :
//   ANTHROPIC_API_KEY (secret) · ACCESS_CODE (secret) · ADMIN_CODE (secret) · ALLOWED_ORIGIN (texte)
// Liaisons (Settings > Bindings) :
//   D1 database -> DB · Workers AI -> AI · KV namespace -> PHOTOS

const MODEL = 'claude-sonnet-5-5';
const MAX_IMAGES = 4;
const MAX_IMAGE_B64 = 4 * 1024 * 1024;
const MAX_AUDIO_B64 = 6 * 1024 * 1024;
const MAX_THUMB = 60000;
const WHISPER = '@cf/openai/whisper-large-v3-turbo';
const VOCAB = "Station de ski, remontées mécaniques, télésiège, téléski, télécabine, gare de départ, gare d'arrivée, dameuse, treuil, pisteur, enneigeur, neige de culture, piste, balisage, atelier, EPI, DUERP, presque accident, situation dangereuse.";
const STATUTS = ['Ouvert', 'En cours', 'Clôturé'];
const NIVEAUX = ['Suppression','Substitution','Protection collective','Organisationnel','EPI','Information / formation'];
const FAMILLES = ['Chute de hauteur','Chute de plain-pied','Manutention / TMS','Circulation / engins','Machines et équipements','Électrique','Incendie / explosion','Chimique','Ambiances climatiques (froid, UV, altitude)','Milieu montagne (avalanche, chute de pierres)','Bruit / vibrations','Co-activité / entreprise extérieure','Organisation / psychosocial','Autre'];
const ID_RE = /^[a-z0-9]{6,40}$/;

const LIST_FIELDS = 'id, date, declarant, declarant_id, type, secteur, lieu, description, note_vocale, statut, thumb, analysis, analysis_status, suivi, nb_photos, has_audio, updated_at, closed_at';
const MINE_FIELDS = 'id, date, type, secteur, lieu, description, note_vocale, statut, thumb, suivi, nb_photos, has_audio, updated_at, closed_at';

export default {
  async fetch(req, env) {
    const origin = req.headers.get('Origin') || '';
    const allowed = (env.ALLOWED_ORIGIN || '').split(',').map(s => s.trim()).filter(Boolean);
    const cors = {
      'Access-Control-Allow-Origin': allowed.includes(origin) ? origin : (allowed[0] || ''),
      'Access-Control-Allow-Methods': 'GET, POST, PATCH, DELETE, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type, X-Access-Code, X-User-Id',
      'Access-Control-Max-Age': '86400',
      'Vary': 'Origin',
    };
    const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...cors, 'Content-Type': 'application/json; charset=utf-8' } });

    if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors });
    if (allowed.length && origin && !allowed.includes(origin)) return json({ error: 'Origine non autorisée.' }, 403);

    const code = req.headers.get('X-Access-Code') || '';
    const role = code && code === env.ADMIN_CODE ? 'admin' : code && code === env.ACCESS_CODE ? 'user' : null;
    if (!role) return json({ error: "Code d'accès incorrect." }, 401);
    const isAdmin = role === 'admin';

    const path = new URL(req.url).pathname.replace(/\/+$/, '');

    try {
      if (path === '/api/ping' && req.method === 'GET') return json({ ok: true, role });

      // ---------- Envoi d'une remontée (tous) ----------
      if (path === '/api/submit' && req.method === 'POST') {
        const b = await req.json();
        if (!b || typeof b.id !== 'string' || !ID_RE.test(b.id)) return json({ error: 'Identifiant invalide.' }, 400);
        const exists = await env.DB.prepare('SELECT id FROM remontees WHERE id = ?').bind(b.id).first();
        if (exists) return json({ ok: true, duplicate: true });

        const images = Array.isArray(b.images) ? b.images.filter(i => typeof i === 'string').slice(0, MAX_IMAGES) : [];
        if (images.some(i => i.length > MAX_IMAGE_B64)) return json({ error: 'Photo trop lourde.' }, 413);
        const audio = b.audio && typeof b.audio.data === 'string' && b.audio.data.length ? b.audio : null;
        if (audio && audio.data.length > MAX_AUDIO_B64) return json({ error: 'Note vocale trop longue.' }, 413);

        const d = {
          type: str(b.type, 50), secteur: str(b.secteur, 100), lieu: str(b.lieu, 300), desc: str(b.desc, 3000),
        };
        if (!images.length && !d.desc && !audio) return json({ error: 'Remontée vide.' }, 400);

        // 1. Conserver photos et audio
        await Promise.all(images.map((data, i) => env.PHOTOS.put(`photo:${b.id}:${i}`, data)));
        if (audio) await env.PHOTOS.put(`audio:${b.id}`, b64ToBytes(audio.data), { metadata: { mime: str(audio.mime, 50) || 'audio/webm' } });

        // 2. Transcription (en cas d'échec, la remontée est quand même enregistrée)
        let transcript = '';
        if (audio) transcript = await transcribe(env, audio.data);

        // 3. Analyse IA (en cas d'échec, l'administrateur pourra la relancer)
        let analysis = {}, status = 'ok';
        try { analysis = await analyze(env, { ...d, transcript, audioFailed: audio && !transcript }, images); }
        catch (err) { console.log('Analyse impossible', err && err.message); status = 'erreur'; }

        const thumb = typeof b.thumb === 'string' && b.thumb.startsWith('data:image/jpeg;base64,') && b.thumb.length < MAX_THUMB ? b.thumb : null;
        const now = new Date().toISOString();
        await env.DB.prepare(`INSERT INTO remontees (id, date, declarant, declarant_id, type, secteur, lieu, description, note_vocale, statut, thumb, analysis, analysis_status, suivi, nb_photos, has_audio, updated_at)
          VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`)
          .bind(b.id, str(b.date, 40) || now, str(b.declarant, 100), uid(b.declarant_id), d.type, d.secteur, d.lieu, d.desc,
            audio && !transcript ? '(transcription impossible, écouter la note)' : transcript.slice(0, 6000),
            'Ouvert', thumb, JSON.stringify(analysis), status, JSON.stringify({ actions: [], message: '' }), images.length, audio ? 1 : 0, now)
          .run();
        return json({ ok: true }, 201);
      }

      // ---------- Mes envois (tous) ----------
      if (path === '/api/mine' && req.method === 'GET') {
        const id = uid(req.headers.get('X-User-Id'));
        if (!id) return json({ items: [] });
        const { results } = await env.DB.prepare(`SELECT ${MINE_FIELDS} FROM remontees WHERE declarant_id = ? ORDER BY date DESC LIMIT 200`).bind(id).all();
        return json({ items: results.map(r => ({ ...r, desc: r.description, suivi: safeParse(r.suivi, { actions: [], message: '' }) })) });
      }

      // ---------- Photos et note vocale de mes envois (tous) ----------
      const mm = path.match(/^\/api\/mine\/([a-z0-9]{6,40})\/(photos|audio)$/);
      if (mm && req.method === 'GET') {
        const me = uid(req.headers.get('X-User-Id'));
        const row = me && await env.DB.prepare('SELECT id, nb_photos, has_audio FROM remontees WHERE id = ? AND declarant_id = ?').bind(mm[1], me).first();
        if (!row) return json({ error: 'Remontée introuvable.' }, 404);
        if (mm[2] === 'photos') return json({ photos: await getPhotos(env, row) });
        return (await audioResponse(env, row.id, cors)) || json({ error: 'Pas de note vocale.' }, 404);
      }

      // ---------- Tout le reste : administrateur uniquement ----------
      if (!isAdmin) return json({ error: 'Réservé aux administrateurs.' }, 403);

      if (path === '/api/remontees' && req.method === 'GET') {
        const { results } = await env.DB.prepare(`SELECT ${LIST_FIELDS} FROM remontees ORDER BY date DESC LIMIT 1000`).all();
        return json({ items: results.map(r => ({ ...r, desc: r.description, analysis: safeParse(r.analysis, {}), suivi: safeParse(r.suivi, { actions: [], message: '' }) })) });
      }

      const m = path.match(/^\/api\/remontees\/([a-z0-9]{6,40})(?:\/(photos|audio|reanalyze))?$/);
      if (m) {
        const [, id, sub] = m;
        const row = await env.DB.prepare('SELECT * FROM remontees WHERE id = ?').bind(id).first();
        if (!row) return json({ error: 'Remontée introuvable.' }, 404);

        if (sub === 'photos' && req.method === 'GET') return json({ photos: await getPhotos(env, row) });
        if (sub === 'audio' && req.method === 'GET') return (await audioResponse(env, id, cors)) || json({ error: 'Pas de note vocale.' }, 404);
        if (sub === 'reanalyze' && req.method === 'POST') {
          const photos = await getPhotos(env, row);
          let transcript = row.note_vocale || '';
          if (row.has_audio && (!transcript || transcript.startsWith('(transcription impossible'))) {
            const a = await env.PHOTOS.get(`audio:${id}`, { type: 'arrayBuffer' });
            if (a) { const t = await transcribe(env, bytesToB64(new Uint8Array(a))); if (t) transcript = t; }
          }
          let analysis;
          try { analysis = await analyze(env, { type: row.type, secteur: row.secteur, lieu: row.lieu, desc: row.description, transcript }, photos); }
          catch (err) { return json({ error: "L'analyse a encore échoué. Réessayez plus tard." }, 502); }
          const now = new Date().toISOString();
          await env.DB.prepare("UPDATE remontees SET analysis = ?, analysis_status = 'ok', note_vocale = ?, updated_at = ? WHERE id = ?")
            .bind(JSON.stringify(analysis), transcript.slice(0, 6000), now, id).run();
          return json({ ok: true, analysis, note_vocale: transcript });
        }
        if (!sub && req.method === 'PATCH') {
          const p = await req.json();
          const sets = [], vals = [];
          const now = new Date().toISOString();
          if (p.statut !== undefined) {
            if (!STATUTS.includes(p.statut)) return json({ error: 'Statut invalide.' }, 400);
            sets.push('statut = ?'); vals.push(p.statut);
            sets.push('closed_at = ?'); vals.push(p.statut === 'Clôturé' ? (row.statut === 'Clôturé' ? row.closed_at : now) : null);
          }
          if (p.analysis !== undefined) {
            const a = JSON.stringify({ ...(p.analysis || {}), modifie_par_admin: true, modifie_le: now });
            if (a.length > 100000) return json({ error: 'Analyse trop volumineuse.' }, 413);
            sets.push('analysis = ?', "analysis_status = 'ok'"); vals.push(a);
          }
          if (p.suivi !== undefined) {
            const s = p.suivi || {};
            const suivi = {
              message: str(s.message, 2000),
              actions: (Array.isArray(s.actions) ? s.actions : []).slice(0, 30).map(x => ({ texte: str(x.texte, 500), fait: !!x.fait })).filter(x => x.texte),
              maj: now,
            };
            sets.push('suivi = ?'); vals.push(JSON.stringify(suivi));
          }
          if (!sets.length) return json({ error: 'Rien à modifier.' }, 400);
          sets.push('updated_at = ?'); vals.push(now);
          await env.DB.prepare(`UPDATE remontees SET ${sets.join(', ')} WHERE id = ?`).bind(...vals, id).run();
          const upd = await env.DB.prepare(`SELECT ${LIST_FIELDS} FROM remontees WHERE id = ?`).bind(id).first();
          return json({ ok: true, item: { ...upd, desc: upd.description, analysis: safeParse(upd.analysis, {}), suivi: safeParse(upd.suivi, { actions: [], message: '' }) } });
        }
        if (!sub && req.method === 'DELETE') {
          await env.DB.prepare('DELETE FROM remontees WHERE id = ?').bind(id).run();
          await Promise.all([...Array.from({ length: row.nb_photos || 0 }, (_, i) => env.PHOTOS.delete(`photo:${id}:${i}`)), env.PHOTOS.delete(`audio:${id}`)]);
          return json({ ok: true });
        }
      }

      return json({ error: 'Introuvable.' }, 404);
    } catch (err) {
      console.log('Erreur serveur', err && err.stack || err);
      return json({ error: 'Erreur serveur.' }, 500);
    }
  },
};

// ---------- Outils ----------
async function getPhotos(env, row) {
  return (await Promise.all(Array.from({ length: row.nb_photos || 0 }, (_, i) => env.PHOTOS.get(`photo:${row.id}:${i}`)))).filter(Boolean);
}
async function audioResponse(env, id, cors) {
  const a = await env.PHOTOS.getWithMetadata(`audio:${id}`, { type: 'arrayBuffer' });
  if (!a || !a.value) return null;
  return new Response(a.value, { headers: { ...cors, 'Content-Type': a.metadata?.mime || 'audio/webm' } });
}
const str = (v, max) => (typeof v === 'string' ? v.trim().slice(0, max) : '');
const uid = v => (typeof v === 'string' && /^[a-z0-9]{8,40}$/.test(v) ? v : null);
const safeParse = (t, def) => { try { return t ? JSON.parse(t) : def; } catch { return def; } };
function b64ToBytes(b64) { const bin = atob(b64); const u = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) u[i] = bin.charCodeAt(i); return u; }
function bytesToB64(u) { let s = ''; for (let i = 0; i < u.length; i += 0x8000) s += String.fromCharCode.apply(null, u.subarray(i, i + 0x8000)); return btoa(s); }
function parseJson(text) {
  try { return JSON.parse(text); } catch {}
  const fence = text.match(/```(?:json)?\s*([\s\S]*?)```/);
  if (fence) { try { return JSON.parse(fence[1]); } catch {} }
  const a = text.indexOf('{'), b = text.lastIndexOf('}');
  if (a >= 0 && b > a) { try { return JSON.parse(text.slice(a, b + 1)); } catch {} }
  return null;
}

async function transcribe(env, b64) {
  try {
    const t = await env.AI.run(WHISPER, { audio: b64, language: 'fr', initial_prompt: VOCAB });
    return String(t?.text || '').trim();
  } catch (err) {
    console.log('Whisper error', err && err.message);
    return '';
  }
}

async function analyze(env, d, images) {
  const content = [
    ...images.map(data => ({ type: 'image', source: { type: 'base64', media_type: 'image/jpeg', data } })),
    { type: 'text', text: buildPrompt({ ...d, nPhotos: images.length }) },
  ];
  const r = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: { 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', 'content-type': 'application/json' },
    body: JSON.stringify({ model: MODEL, max_tokens: 4000, messages: [{ role: 'user', content }] }),
  });
  if (!r.ok) throw new Error(`Anthropic ${r.status}: ${(await r.text()).slice(0, 300)}`);
  const out = await r.json();
  const text = (out.content || []).filter(c => c.type === 'text').map(c => c.text).join('\n');
  const analysis = parseJson(text);
  if (!analysis) throw new Error('Réponse illisible');
  return analysis;
}

function buildPrompt(d) {
  return `Tu es préventeur HSE expérimenté dans une grande station de ski française (domaines de haute montagne : remontées mécaniques, damage, pistes, ateliers, neige de culture, restauration, activités d'été). Tu analyses une remontée terrain faite par un salarié.

Type déclaré : ${d.type}
Secteur : ${d.secteur}
Lieu : ${d.lieu || 'non précisé'}
Description écrite du déclarant : ${d.desc || 'aucune'}
Note vocale du déclarant (transcription automatique, peut contenir des erreurs de reconnaissance à interpréter avec bon sens, ex. "dame use" pour "dameuse") : ${d.transcript || (d.audioFailed ? 'note vocale présente mais non transcrite' : 'aucune')}
Nombre de photos jointes : ${d.nPhotos}

Consignes :
- Base-toi uniquement sur ce qui est visible sur les photos, écrit dans la description et dit dans la note vocale. Si la note vocale apporte des faits (lieu, circonstances, personnes exposées), intègre-les. N'invente pas de faits. Si un élément n'est pas visible ou pas certain, mets-le dans "points_a_verifier".
- Identifie de 1 à 4 dangers, du plus critique au moins critique.
- "famille" doit être exactement l'une de : ${FAMILLES.join(' | ')}.
- Gravité (1 à 4) : 1 bénigne (soins sans arrêt), 2 sérieuse (arrêt de travail), 3 grave (séquelles, incapacité permanente), 4 très grave (décès ou invalidité lourde).
- Probabilité (1 à 4) : 1 improbable, 2 rare, 3 probable, 4 fréquente. Tiens compte de l'exposition réelle (fréquentation, saison, nombre de personnes exposées).
- Mesures préventives classées selon les principes généraux de prévention (article L4121-2 du Code du travail). "niveau" doit être exactement l'un de : ${NIVEAUX.join(' | ')}. Propose d'abord les niveaux les plus élevés quand c'est réaliste. Mesures concrètes et applicables en station.
- "responsable_suggere" : une fonction (ex. chef d'exploitation RM, responsable atelier), jamais un nom.
- "references" : seulement des textes réglementaires ou normes dont tu es certain qu'ils s'appliquent, sinon liste vide.
- Si le type déclaré semble incorrect (ex. il y a eu un événement alors que c'est déclaré en situation dangereuse, ou il y a eu un blessé), indique-le dans "qualification_commentaire", sinon chaîne vide.
- Rédige en français, phrases courtes, ton factuel et non culpabilisant.

Réponds uniquement avec un objet JSON de cette forme, sans texte autour :
{"synthese":"1 à 2 phrases","qualification_commentaire":"","photo_exploitable":true,"remarque_photo":"","dangers":[{"famille":"","danger":"","dommage_possible":"","gravite":1,"probabilite":1,"justification":""}],"mesures_immediates":[""],"mesures_preventives":[{"niveau":"","mesure":"","responsable_suggere":""}],"points_a_verifier":[""],"references":[""]}`;
}
